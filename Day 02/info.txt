# PASS-THROUGH RERANKING
# ---------------------------------------------------------------

def cross_encoder_rerank(
    query: str,
    candidates: list,
    top_k: int = 5
) -> list:
    """
    Pass through the candidates without using a cross-encoder.

    Keeps the same function interface so the next steps
    in the pipeline continue to work.

    Args:
        query      : original user question (kept for compatibility).
        candidates : list of (Document, score) tuples.
        top_k      : how many documents to return.

    Returns:
        List of (Document, score) tuples.
    """
    if not candidates:
        return []

    return candidates[:top_k]
